# 📦 Полный гайд для демоэкзамена: вёрстка + PostgreSQL + FastAPI

Ниже — единый файл со всем необходимым: список библиотек, шаблоны HTML/CSS, шаблоны PostgreSQL, шаблоны Python (FastAPI), команды запуска виртуального окружения и сервера. Всё максимально прокомментировано, чтобы ты мог адаптировать под любую другую тематику задания.

## 1. 📚 Список библиотек, которые нужно установить

### Обязательные (базовые)
```bash
pip install fastapi
pip install uvicorn[standard]
pip install jinja2
pip install python-multipart
pip install psycopg2-binary
pip install sqlalchemy
pip install pydantic
pip install email-validator
pip install passlib[bcrypt]
pip install python-dotenv
```

### Что за что отвечает
| Библиотека | Назначение |
|---|---|
| **fastapi** | Веб-фреймворк (аналог Flask, но современнее) |
| **uvicorn[standard]** | ASGI-сервер, на котором запускается FastAPI |
| **jinja2** | Шаблонизатор HTML (как в Flask) |
| **python-multipart** | Обработка форм (POST-запросы) |
| **psycopg2-binary** | Драйвер для подключения к PostgreSQL |
| **sqlalchemy** | ORM для работы с БД (можно и без неё, но удобнее) |
| **pydantic** | Валидация данных (встроена в FastAPI) |
| **email-validator** | Проверка email |
| **passlib[bcrypt]** | Хеширование паролей |
| **python-dotenv** | Загрузка переменных окружения из `.env` |

### Для фронтенда (если нужен)
- **Bootstrap 5** — подключается через CDN, скачивать не нужно
- **jQuery** — необязательно, для валидации на клиенте

-- Тематика: сервис заявок на услуги

-- ============================================================

DROP TABLE IF EXISTS services CASCADE;

-- ------------------------------------------------------------

-- ------------------------------------------------------------

    id            SERIAL PRIMARY KEY,                -- уникальный ID

-- Справочник услуг (виды услуг для выпадающего списка)
-- ------------------------------------------------------------
CREATE TABLE services (
    id    SERIAL PRIMARY KEY,

);

-- Начальные услуги (можно менять под тематику)

## 4. 🐍 Шаблон FastAPI (`main.py`) с комментариями

```python
# ============================================================
# ШАБЛОН FastAPI-ПРИЛОЖЕНИЯ ДЛЯ ДЕМОЭКЗАМЕНА
# Меняй модели/маршруты под свою тематику
# ============================================================

import os
import re
from datetime import date, time, datetime
from typing import Optional

from fastapi import FastAPI, Request, Form, Depends, HTTPException, status
from fastapi.responses import HTMLResponse, RedirectResponse
from fastapi.templating import Jinja2Templates
from fastapi.staticfiles import StaticFiles
from sqlalchemy import create_engine, Column, Integer, String, Text, Date, Time, ForeignKey, TIMESTAMP, func
from sqlalchemy.orm import declarative_base, sessionmaker, Session, relationship
from passlib.hash import bcrypt
from dotenv import load_dotenv

# ------------------------------------------------------------
# 1. ЗАГРУЗКА ПЕРЕМЕННЫХ ОКРУЖЕНИЯ
# ------------------------------------------------------------
load_dotenv()

DB_HOST = os.getenv("DB_HOST", "localhost")
DB_PORT = os.getenv("DB_PORT", "5432")
DB_NAME = os.getenv("DB_NAME", "demo_exam")
DB_USER = os.getenv("DB_USER", "postgres")
DB_PASSWORD = os.getenv("DB_PASSWORD", "postgres")

DATABASE_URL = f"postgresql+psycopg2://{DB_USER}:{DB_PASSWORD}@{DB_HOST}:{DB_PORT}/{DB_NAME}"

# ------------------------------------------------------------
# 2. ПОДКЛЮЧЕНИЕ К БД (SQLAlchemy)
# ------------------------------------------------------------
engine = create_engine(DATABASE_URL, echo=False)
SessionLocal = sessionmaker(bind=engine, autoflush=False, autocommit=False)
Base = declarative_base()

# ------------------------------------------------------------
# 3. МОДЕЛИ (соответствуют таблицам в SQL)
# ------------------------------------------------------------
class User(Base):
    __tablename__ = "users"
    id            = Column(Integer, primary_key=True)
    login         = Column(String(50), unique=True, nullable=False)
    password_hash = Column(String(255), nullable=False)
    full_name     = Column(String(150), nullable=False)
    phone         = Column(String(20), nullable=False)
    email         = Column(String(100), nullable=False)
    role          = Column(String(20), default="user")
    created_at    = Column(TIMESTAMP, server_default=func.now())

class Service(Base):
    __tablename__ = "services"
    id   = Column(Integer, primary_key=True)
    name = Column(String(100), unique=True, nullable=False)

class RequestModel(Base):
    __tablename__ = "requests"
    id             = Column(Integer, primary_key=True)
    user_id        = Column(Integer, ForeignKey("users.id", ondelete="CASCADE"), nullable=False)
    service_id     = Column(Integer, ForeignKey("services.id"), nullable=True)
    custom_service = Column(Text, nullable=True)
    address        = Column(String(255), nullable=False)
    phone          = Column(String(20), nullable=False)
    desired_date   = Column(Date, nullable=False)
    desired_time   = Column(Time, nullable=False)
    payment_type   = Column(String(20), nullable=False)
    status         = Column(String(20), default="новая")
    cancel_reason  = Column(Text, nullable=True)
    created_at     = Column(TIMESTAMP, server_default=func.now())

    user    = relationship("User")
    service = relationship("Service")

# Создание таблиц, если их нет (обычно делается через SQL-скрипт, но пусть будет)
Base.metadata.create_all(bind=engine)

# ------------------------------------------------------------
# 4. ИНИЦИАЛИЗАЦИЯ ПРИЛОЖЕНИЯ
# ------------------------------------------------------------
app = FastAPI(title="Демоэкзамен")
templates = Jinja2Templates(directory="templates")
app.mount("/static", StaticFiles(directory="static"), name="static")

# ------------------------------------------------------------
# 5. ЗАВИСИМОСТЬ: получение сессии БД
# ------------------------------------------------------------
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

# ------------------------------------------------------------
# 6. ВАЛИДАТОРЫ (регулярки под задание)
# ------------------------------------------------------------
PHONE_RE = re.compile(r"^\+7\(\d{3}\)-\d{3}-\d{2}-\d{2}$")
FIO_RE   = re.compile(r"^[А-Яа-яЁё\s]+$")
EMAIL_RE = re.compile(r"^[^@\s]+@[^@\s]+\.[^@\s]+$")

def validate_registration(login, password, full_name, phone, email):
    """Возвращает список ошибок (пустой = всё ок)."""
    errors = []
    if not login or len(login) < 3:
        errors.append("Логин должен быть не менее 3 символов")
    if not password or len(password) < 6:
        errors.append("Пароль должен быть не менее 6 символов")
    if not FIO_RE.match(full_name or ""):
        errors.append("ФИО должно содержать только кириллицу и пробелы")
    if not PHONE_RE.match(phone or ""):
        errors.append("Телефон должен быть в формате +7(XXX)-XXX-XX-XX")
    if not EMAIL_RE.match(email or ""):
        errors.append("Некорректный email")
    return errors

# ------------------------------------------------------------
# 7. ХЕЛПЕР: текущий пользователь из cookie (простая сессия)
# ------------------------------------------------------------
def get_current_user(request: Request, db: Session):
    user_id = request.cookies.get("user_id")
    if not user_id:
        return None
    return db.query(User).filter(User.id == int(user_id)).first()

# ------------------------------------------------------------
# 8. МАРШРУТЫ
# ------------------------------------------------------------

# ---------- Главная ----------
@app.get("/", response_class=HTMLResponse)
def index(request: Request, db: Session = Depends(get_db)):
    user = get_current_user(request, db)
    return templates.TemplateResponse("index.html", {"request": request, "user": user})

# ---------- Регистрация ----------
@app.get("/register", response_class=HTMLResponse)
def register_form(request: Request):
    return templates.TemplateResponse("register.html", {"request": request, "errors": []})

@app.post("/register", response_class=HTMLResponse)
def register(
    request: Request,
    login: str = Form(...),
    password: str = Form(...),
    full_name: str = Form(...),
    phone: str = Form(...),
    email: str = Form(...),
    db: Session = Depends(get_db),
):
    errors = validate_registration(login, password, full_name, phone, email)

    # Проверка уникальности логина
    if db.query(User).filter(User.login == login).first():
        errors.append("Такой логин уже занят")

    if errors:
        return templates.TemplateResponse(
            "register.html",
            {"request": request, "errors": errors, "form": {
                "login": login, "full_name": full_name,
                "phone": phone, "email": email
            }},
        )

    user = User(
        login=login,
        password_hash=bcrypt.hash(password),
        full_name=full_name,
        phone=phone,
        email=email,
    )
    db.add(user)
    db.commit()
    return RedirectResponse("/login", status_code=303)

# ---------- Авторизация ----------
@app.get("/login", response_class=HTMLResponse)
def login_form(request: Request):
    return templates.TemplateResponse("login.html", {"request": request, "error": None})

@app.post("/login", response_class=HTMLResponse)
def login(
    request: Request,
    login: str = Form(...),
    password: str = Form(...),
    db: Session = Depends(get_db),
):
    user = db.query(User).filter(User.login == login).first()
    if not user or not bcrypt.verify(password, user.password_hash):
        return templates.TemplateResponse(
            "login.html",
            {"request": request, "error": "Неверный логин или пароль"},
        )

    # Простая сессия через cookie
    redirect = RedirectResponse("/", status_code=303)
    redirect.set_cookie("user_id", str(user.id), httponly=True)
    return redirect

@app.get("/logout")
def logout():
    redirect = RedirectResponse("/", status_code=303)
    redirect.delete_cookie("user_id")
    return redirect

# ---------- Создание заявки (история + форма) ----------
@app.get("/requests", response_class=HTMLResponse)
def requests_page(request: Request, db: Session = Depends(get_db)):
    user = get_current_user(request, db)
    if not user:
        return RedirectResponse("/login", status_code=303)

    my_requests = (
        db.query(RequestModel)
        .filter(RequestModel.user_id == user.id)
        .order_by(RequestModel.created_at.desc())
        .all()
    )
    services = db.query(Service).all()
    return templates.TemplateResponse(
        "requests.html",
        {"request": request, "user": user, "requests": my_requests, "services": services},
    )

@app.post("/requests")
def create_request(
    request: Request,
    address: str = Form(...),
    phone: str = Form(...),
    desired_date: str = Form(...),
    desired_time: str = Form(...),
    service_id: Optional[str] = Form(None),
    custom_service: Optional[str] = Form(None),
    payment_type: str = Form(...),
    db: Session = Depends(get_db),
):
    user = get_current_user(request, db)
    if not user:
        return RedirectResponse("/login", status_code=303)

    # Валидация
    errors = []
    if not address.strip():
        errors.append("Укажите адрес")
    if not PHONE_RE.match(phone):
        errors.append("Телефон в формате +7(XXX)-XXX-XX-XX")
    if not service_id and not (custom_service and custom_service.strip()):
        errors.append("Выберите услугу или опишите свою")
    if payment_type not in ("cash", "card"):
        errors.append("Выберите тип оплаты")

    if errors:
        services = db.query(Service).all()
        my_requests = db.query(RequestModel).filter(RequestModel.user_id == user.id).all()
        return templates.TemplateResponse(
            "requests.html",
            {"request": request, "user": user, "requests": my_requests,
             "services": services, "errors": errors},
        )

    req = RequestModel(
        user_id=user.id,
        service_id=int(service_id) if service_id else None,
        custom_service=custom_service if not service_id else None,
        address=address,
        phone=phone,
        desired_date=date.fromisoformat(desired_date),
        desired_time=time.fromisoformat(desired_time),
        payment_type=payment_type,
    )
    db.add(req)
    db.commit()
    return RedirectResponse("/requests", status_code=303)

# ---------- Панель администратора ----------
@app.get("/admin", response_class=HTMLResponse)
def admin_panel(request: Request, db: Session = Depends(get_db)):
    user = get_current_user(request, db)
    if not user or user.role != "admin":
        raise HTTPException(status_code=403, detail="Доступ запрещён")

    all_requests = (
        db.query(RequestModel)
        .order_by(RequestModel.created_at.desc())
        .all()
    )
    return templates.TemplateResponse(
        "admin.html",
        {"request": request, "user": user, "requests": all_requests},
    )

@app.post("/admin/status")
def admin_change_status(
    request: Request,
    request_id: int = Form(...),
    new_status: str = Form(...),
    cancel_reason: Optional[str] = Form(None),
    db: Session = Depends(get_db),
):
    user = get_current_user(request, db)
    if not user or user.role != "admin":
        raise HTTPException(status_code=403)

    req = db.query(RequestModel).filter(RequestModel.id == request_id).first()
    if not req:
        raise HTTPException(status_code=404)

    req.status = new_status
    if new_status == "отменено":
        req.cancel_reason = cancel_reason or "Причина не указана"
    db.commit()
    return RedirectResponse("/admin", status_code=303)
```

## 8. 🚀 Порядок запуска

1. **Создать БД в PostgreSQL:**
   ```bash
   psql -U postgres -c "CREATE DATABASE demo_exam;"
   psql -U postgres -d demo_exam -f database.sql
   ```

2. **Создать и активировать виртуальное окружение:**
   ```bash
   python -m venv venv
   venv\Scripts\activate          # Windows
   source venv/bin/activate       # Linux/macOS
   ```

3. **Установить зависимости:**
   ```bash
   pip install -r requirements.txt
   ```

4. **Запустить сервер:**
   ```bash
   uvicorn main:app --reload
   ```

5. **Открыть в браузере:** `http://127.0.0.1:8000`

## 💡 Как адаптировать под другую тематику

- **Модели** (`User`, `Service`, `RequestModel`) — переименуй поля под свою предметную область (например, `Book`, `Order`, `Ticket`).
- **SQL-скрипт** — замени таблицы и связи, оставь структуру: справочник + основная таблица + пользователи.
- **Валидаторы** — регулярки под конкретные форматы (ИНН, паспорт, артикул и т.д.).
